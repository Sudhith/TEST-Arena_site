Benchmark Harness Quick Reference (CLI) 
# ── Digit CAPTCHA Pipeline ──────────────────────────────────────
# 1. Fetch Challenge
curl -s http://localhost:8000/api/captcha-digit

# 2. Verify answer with Ground Truth Returned
curl -s -X POST http://localhost:8000/api/verify-digit \
  -H "Content-Type: application/json" \
  -d '{"session_id": "<SESSION_ID>", "answer": "580608"}'

# ── Grid Selection Pipeline ─────────────────────────────────────
# 1. Fetch Challenge
curl -s http://localhost:8000/api/captcha-grid

# 2. Verify Selected Tile Indices
curl -s -X POST http://localhost:8000/api/verify-grid \
  -H "Content-Type: application/json" \
  -d '{"session_id": "<SESSION_ID>", "selected_indices": [0, 3, 7]}'
