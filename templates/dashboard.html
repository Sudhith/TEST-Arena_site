Agent Integration Quick Reference

# 1. Fetch a digit CAPTCHA
curl http://localhost:8000/api/captcha-digit

# 2. Verify your answer
curl -X POST http://localhost:8000/api/verify-digit \
  -H "Content-Type: application/json" \
  -d '{"session_id": "...", "answer": "384729"}'

# 3. Fetch a grid CAPTCHA
curl http://localhost:8000/api/captcha-grid

# 4. Verify selected tile indices
curl -X POST http://localhost:8000/api/verify-grid \
  -H "Content-Type: application/json" \
  -d '{"session_id": "...", "selected_indices": [0, 3, 7]}'
