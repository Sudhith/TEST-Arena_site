bash • curl automated pipeline 
# 1. Fetch Challenge
CHALLENGE=$(curl -s http://localhost:8000/api/captcha-digit)
SESSION_ID=$(echo $CHALLENGE | jq -r '.session_id')

# 2. Download Image (feed into your solver model)
curl -s http://localhost:8000/captcha-image/$SESSION_ID -o captcha.png

# 3. Submit Predicted 6-Digit String & Receive Ground-Truth Signal

  -H "Content-Type: application/json" \
  -d "{\"session_id\": \"$SESSION_ID\", \"answer\": \"384729\"}"
python • requests OCR benchmarking pipeline 
import requests

BASE_URL = "http://localhost:8000"

# 1. Request new challenge from testbed
resp = requests.get(f"{BASE_URL}/api/captcha-digit").json()
session_id = resp["session_id"]
image_bytes = requests.get(f"{BASE_URL}{resp['captcha_image_url']}").content

# 2. Run your OCR/Vision Model (Replace with model prediction)
predicted_text = "580608"

# 3. Submit and receive immediate training reward signal
verify_resp = requests.post(
    f"{BASE_URL}/api/verify-digit",
    json={"session_id": session_id, "answer": predicted_text}
).json()

print("Outcome:", verify_resp["success"])
print("Ground truth:", verify_resp["correct_answer"])
javascript • browser / node fetch pipeline 
async function benchmarkSolver() {
  // 1. Fetch live 3x3 Grid Challenge
  const res = await fetch("http://localhost:8000/api/captcha-grid");
  const challenge = await res.json();
  console.log("Task:", challenge.instruction);

  // 2. Submit AI Agent predicted tile indices
  const verifyRes = await fetch("http://localhost:8000/api/verify-grid", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      session_id: challenge.session_id,
      selected_indices: [0, 3, 7]
    })
  });
  const result = await verifyRes.json();
  console.log("Result:", result);
}
